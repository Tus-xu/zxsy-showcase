import { ApiError, api } from "/src/api/client.js";
import { redirectTeacherDenied } from "/src/api/teacher/client.ts";
function guard(e) {
  if (e instanceof ApiError && e.code === 40301) redirectTeacherDenied();
  throw e;
}
async function raw(method, path, opts = {}) {
  const r = await api.raw(method, path, opts);
  return r.data;
}
export const api8000 = {
  /* ---------- 班级（8000 /api/classes*，身份与成员唯一权威） ---------- */
  classes: {
    list: () => raw("GET", "/classes/mine").catch(guard),
    create: (body) => raw("POST", "/classes", { body }).catch(guard),
    members: (classId) => raw("GET", `/classes/${classId}/members`).catch(guard),
    /** 教师确认/取消确认成员（pending → confirmed 才进任务/学情口径，D3） */
    setMemberConfirmed: (classId, userId, confirmed) => raw("PATCH", `/classes/${classId}/members/${userId}`, { body: { confirmed } }).catch(guard),
    resetInviteCode: (classId) => raw("POST", `/classes/${classId}/invite-code/reset`).catch(guard)
  },

    confirm: async (artifactId, clientRequestId, idempotencyKey) => {
      const r = await raw(
        "POST",
        `/teacher/artifacts/${artifactId}/confirm`,
        { body: { client_request_id: clientRequestId }, idempotencyKey }
      ).catch(guard);
      const art = r.artifact ?? r;
      return Object.assign(art, { replayed: r.replayed });
    }
  },

      body: { client_request_id: clientRequestId },
      idempotencyKey
    }).catch(guard),

    }).catch(guard)
