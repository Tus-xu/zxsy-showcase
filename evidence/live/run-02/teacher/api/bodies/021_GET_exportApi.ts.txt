export const ARTIFACTS_API_BASE = "/api/teacher-v3/artifacts";
function authHeader() {
  const token = getAccessToken();
  return token ? { Authorization: `Bearer ${token}` } : {};
}
function kindOfStatus(status) {
  if (status === 401) return "unauthenticated";
  if (status === 403) return "forbidden";
  if (status === 409) return "conflict";
  if (status === 404) return "not_found";
  if (status === 429) return "rate_limited";
  return "error";
}
async function artifactRequest(path, init) {
  let res;
  try {
    res = await fetch(`${ARTIFACTS_API_BASE}${path}`, {
      ...init,
      headers: {
        Accept: "application/json",
        "Content-Type": "application/json",
        ...authHeader(),
        ...init.headers || {}
      }
    });
  } catch (e) {
    if (e?.name === "AbortError") throw e;
    throw new PrepCenterApiError({ kind: "network", status: null, code: null, detail: e?.message });
  }
  const body = await res.json().catch(() => null);
  if (!res.ok || !body || typeof body.code === "number" && body.code !== 0) {
    throw new PrepCenterApiError({
      kind: kindOfStatus(res.status),
      status: res.status,
      code: typeof body?.code === "number" ? body.code : null,
      detail: body?.message || body?.detail || `artifact api error: ${res.status}`,
      requestId: body?.request_id ?? null
    });
  }
  return body.data ?? body;
}

export function formatBytes(n) {
  if (!Number.isFinite(n) || n <= 0) return "";
  if (n < 1024) return `${n} B`;
  const kb = n / 1024;
  if (kb < 1024) return `${kb.toFixed(1)} KB`;
  const mb = kb / 1024;
  if (mb < 1024) return `${mb.toFixed(1)} MB`;
  return `${(mb / 1024).toFixed(2)} GB`;
}

function parseFilename(disposition) {
  if (!disposition) return null;
  const star = /filename\*\s*=\s*([^;]+)/i.exec(disposition);
  if (star) {
    let raw = star[1].trim().replace(/^["']|["']$/g, "");
    const m = /^[^']*'[^']*'(.*)$/.exec(raw);
    if (m) raw = m[1];
    try {
      const decoded = decodeURIComponent(raw);
      if (decoded) return decoded;
    } catch {
      if (raw) return raw;
    }
  }
  const plain = /filename\s*=\s*("([^"]*)"|([^;]+))/i.exec(disposition);
  if (plain) {
    const v = (plain[2] ?? plain[3] ?? "").trim();
    if (v) return v;
  }
  return null;
}
